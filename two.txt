import ollama
import ast

# 1. GENERATOR (Ask CodeGemma)
print("Generating code using CodeGemma...")

# FIX 1: Move """ to the left and add a comment
prompt = """def insertion_sort_function(arr):
    # Write the insertion sort logic here
"""

code = ""
while True:
    response = ollama.chat( 
        model="codegemma:2b", 
        messages=[{"role": "user", "content": prompt}],
        options={"num_predict": 150} 
    ) 

    raw_code = response["message"]["content"]
    cleaned = raw_code.replace("```python", "").replace("```", "").strip()
    start = cleaned.find("def insertion_sort_function")
    
    if start != -1:
        code = cleaned[start:]
        
        # CHOP OFF everything after "return arr"
        end = code.find("return arr")
        if end != -1:
            code = code[:end + 10]
            
            # FIX 2: Lowered to 20 so it doesn't get stuck in an infinite loop
            if len(code) > 20 and "pass" not in code:
                break
                
    print("AI format error. Retrying...")

print("\n--- GENERATED CODE ---")
print(code)
print("----------------------")

# 2. DISCRIMINATOR (Test the code)
print("\nTesting Function...")

# Check 1: Syntax
try:
    ast.parse(code)
    syntax_ok = True
except:
    syntax_ok = False

# Check 2: Algorithm (Did it cheat and use .sort()?)
algo_ok = "sort(" not in code and "sorted(" not in code

# Check 3: Functional Test (Does it actually sort?)
try:
    namespace = {}
    exec(code, namespace)
    func = namespace["insertion_sort_function"]
    test_array = [5, 2, 8, 1]
    result = func(test_array)
    func_ok = (result == [1, 2, 5, 8])
except:
    func_ok = False

# 3. FINAL RESULT
print(f"Syntax Check: {syntax_ok}")
print(f"Algorithm Check (No built-in sort): {algo_ok}")
print(f"Functional Test: {func_ok}")

if syntax_ok and algo_ok and func_ok:
    print("Final Score: 100/100")
    print("RESULT: ACCEPTED")
else:
    print("RESULT: REJECTED")
