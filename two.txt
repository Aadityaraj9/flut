import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Flutter Demo',
      theme: ThemeData(primarySwatch: Colors.blue),
      home: const MyHomePage(title: 'Responsive Dashboard'),
    );
  }
}

class MyHomePage extends StatefulWidget {
  const MyHomePage({super.key, required this.title});

  final String title;

  @override
  State<MyHomePage> createState() => _MyHomePageState();
}

class _MyHomePageState extends State<MyHomePage> {
  // These were for the counter; kept so you don't have to delete the lines.
  int _counter = 0;

  void _incrementCounter() {
    setState(() {
      _counter++;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text(widget.title),
      ),
      body: LayoutBuilder(
        builder: (context, constraints) {
          // Small screen (width < 600) → Column (stacked vertically)
          if (constraints.maxWidth < 600) {
            return const Column(
              children: [
                Card(
                  child: ListTile(
                    leading: Icon(Icons.people),
                    title: Text('Students'),
                  ),
                ),
                Card(
                  child: ListTile(
                    leading: Icon(Icons.book),
                    title: Text('Courses'),
                  ),
                ),
              ],
            );
          }

          // Large screen → Row (side by side, each takes half width)
          return const Row(
            children: [
              Expanded(
                child: Card(
                  child: ListTile(
                    leading: Icon(Icons.people),
                    title: Text('Students'),
                  ),
                ),
              ),
              Expanded(
                child: Card(
                  child: ListTile(
                    leading: Icon(Icons.book),
                    title: Text('Courses'),
                  ),
                ),
              ),
            ],
          );
        },
      ),
      // Replaced floatingActionButton with null so you don't need to delete the block
      floatingActionButton: null,
    );
  }
}
