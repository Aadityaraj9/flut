import ollama

print("Generating code...")
prompt = "Write a Python function named factorial that takes a non-negative integer n. Return ONLY the Python code. No markdown, no explanations."
response = ollama.generate(model="qwen2.5-coder:7b", prompt=prompt)

code = response["response"].replace("```python", "").replace("```", "").strip()
print("\n--- GENERATED CODE ---")
print(code)
print("----------------------\n")

print("Testing Function...")
namespace = {}
exec(code, namespace)
factorial = namespace["factorial"]

test_cases = [(0, 1), (1, 1), (3, 6), (5, 120), (7, 5040), (10, 3628800)]
passed = 0

for n, expected in test_cases:
    result = factorial(n)
    if result == expected:
        print(f"factorial({n}) = {result} (Expected: {expected}) -> PASS")
        passed += 1
    else:
        print(f"factorial({n}) = {result} (Expected: {expected}) -> FAIL")

print("-" * 40)
print(f"Test Cases Passed: {passed}/{len(test_cases)}")
if passed == len(test_cases):
    print("Overall Result: CORRECT")
else:
    print("Overall Result: INCORRECT")
