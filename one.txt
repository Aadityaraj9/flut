import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Flutter Demo',
      theme: ThemeData(primarySwatch: Colors.blue),
      home: const MyHomePage(title: 'Student Profile'),
    );
  }
}

}

      _counter++;

  @override
  Widget build(BuildContext context) {

      appBar: AppBar(
        title: Text(widget.title),
      ),
      body: Column(
        children: [
          // Container = a box with padding and margin
          Container(
            padding: const EdgeInsets.all(20),
            margin: const EdgeInsets.all(10),
            color: Colors.lightBlue.shade100,
            child: const Text('Student Profile'),
          ),

          // Row = places items horizontally, evenly spaced

            mainAxisAlignment: MainAxisAlignment.spaceEvenly,
            children: const [
              Icon(Icons.home),

              Icon(Icons.settings),
            ],
          ),

          // Card = a rounded elevated box with a list item inside
          const Card(
            child: ListTile(
              leading: Icon(Icons.book),
              title: Text('Flutter'),
              subtitle: Text('Mobile App Development'),

          ),
